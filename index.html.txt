        // Получить текущую цену запчасти (с учетом скидки от инструментов)
        function getPartPrice(part) {
            let price = basePartPrices[part];
            if (state.upgrades.screwdriver > 0) {
                price = price * (1 - (0.1 * state.upgrades.screwdriver)); // -10% за каждый уровень
            }
            return Math.floor(price);
        }

        // Генерация телефона
        function generatePhone() {
            if (state.phones.length >= 4) return;
            
            // Фильтруем модели по доступному уровню
            const availableModels = phoneModels.filter(m => m.reqLvl <= state.level);
            const model = availableModels[Math.floor(Math.random() * availableModels.length)];
            
            const isBroken = () => Math.random() > 0.3;
            let parts = { screen: isBroken(), battery: isBroken(), camera: isBroken() };
            if (!parts.screen && !parts.battery && !parts.camera) parts.screen = true;

            const newPhone = {
                id: Date.now() + Math.random(),
                name: model.name,
                icon: model.icon,
                buyPrice: Math.floor(model.basePrice * (0.9 + Math.random() * 0.3)),
                sellPrice: model.sellPrice,
                needStation: model.needStation || false,
                parts: parts,
                isOwned: false
            };

            state.phones.push(newPhone);
            renderPhones();
        }

        // Действия
        function buyPhone(id) {
            const phone = state.phones.find(p => p.id === id);
            if (phone.needStation && !state.upgrades.station) {
                showToast("❌ Нужна паяльная станция!");
                return;
            }
            if (state.balance >= phone.buyPrice) {
                state.balance -= phone.buyPrice;
                phone.isOwned = true;
                updateUI();
                renderPhones();
                showToast("Успешно куплено!");
            } else {
                showToast("❌ Мало денег!");
            }
        }

        function fixPart(id, part) {
            const phone = state.phones.find(p => p.id === id);
            const cost = getPartPrice(part);
            
            if (state.balance >= cost) {
                state.balance -= cost;
                phone.parts[part] = false;
                updateUI();
                renderPhones();
                showToast(`Починено за ${cost} ₽`);
            } else {
                showToast("❌ Нет денег на деталь!");
            }
        }
